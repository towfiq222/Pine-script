getBarTime(int t, int barsOffset) =>
    t + (barsOffset * timeframe.in_seconds(timeframe.period) * 1000)

zoneDistance(float px, float zTop, float zBot) =>
    px > zTop ? px - zTop : px < zBot ? zBot - px : 0.0

method insertBest(array<float> dists, array<int> idxs, float dist, int idx, int kMax) =>
    if dists.size() == 0
        dists.push(dist)
        idxs.push(idx)
    else
        int pos = dists.size()
        if dists.size() > 0
            for j = 0 to dists.size() - 1
                if dist < dists.get(j)
                    pos := j
                    break
        dists.insert(pos, dist)
        idxs.insert(pos, idx)
    while dists.size() > kMax and dists.size() > 0
        dists.pop()
        idxs.pop()

pickNearest(array<fx_OB> store, bool wantBull, int kMax, bool includeInvalid) =>
    array<float> dists = array.new<float>()
    array<int> idxs = array.new<int>()
    if store.size() > 0
        for i = 0 to store.size() - 1
            fx_OB ob = store.get(i)
            if not na(ob) and ob.isValid
                bool ok = includeInvalid ? true : ob.isValid
                if ok
                    float dist = zoneDistance(close, ob.top, ob.bottom)
                    insertBest(dists, idxs, dist, i, kMax)
    idxs

    if obs.size() > 0

hasGapBetween(int anchorIdx, int bosIdx, bool isBull) =>
    bool gap = false
    int fromIdx = math.min(anchorIdx, bosIdx)
    int toIdx = math.max(anchorIdx, bosIdx)
    if toIdx - fromIdx >= 1 and toIdx <= bar_index and fromIdx >= 0
        for absIdx = fromIdx + 1 to toIdx
            int relNow = bar_index - absIdx
            int relPrev = relNow + 1
            if relNow >= 0 and relPrev >= 0 and relPrev <= bar_index and relNow <= bar_index
                float hiPrev = high[relPrev]
                float loPrev = low[relPrev]
                float hiNow = high[relNow]
                float loNow = low[relNow]
                if not na(hiPrev) and not na(loPrev) and not na(hiNow) and not na(loNow)
                    if isBull
                        if loNow > hiPrev
                            gap := true
                            break
                    else
                        if hiNow < loPrev
                            gap := true
                            break
    gap

obOverlapsActive(float zoneTop, float zoneBottom, array<fx_OB> existingZones) =>
    bool overlaps = false
    if not na(existingZones) and array.size(existingZones) > 0
        for i = 0 to array.size(existingZones) - 1
            fx_OB ob = array.get(existingZones, i)
            if not na(ob) and ob.isValid
                float oTop = math.max(ob.top, ob.bottom)
                float oBot = math.min(ob.top, ob.bottom)
                if zoneTop >= oBot and zoneBottom <= oTop
                    overlaps := true
                    break
    overlaps

v1_detect() =>
    var int v1_os = -1
    var float v1_lastBullOB = na
    var float v1_lastBearOB = na
    varip bool v1_bullCond = false
    varip bool v1_bearCond = false
    
    if bar_index < v1_pivotLen
        [float(na), float(na), v1_os, v1_bullCond, v1_bearCond, v1_bullCond, v1_bearCond]
    else
        bool currentPhv = not na(v1_phv_current)
        
        if high[v1_pivotLen] > v1_highest_prev
            v1_os := 0
        if low[v1_pivotLen] < v1_lowest_prev
            v1_os := 1
            
        v1_bullCond := currentPhv and v1_os == 1
        v1_bearCond := currentPhv and v1_os == 0
        
        if v1_bullCond and not na(low[v1_pivotLen])
            v1_lastBullOB := low[v1_pivotLen]
            bool duplicateFound = false
            int obTime = time[v1_pivotLen]
            if not na(obTime)
                for existing in v1_bullObs
                    if not na(existing) and existing.startTime == obTime
                        duplicateFound := true
                        break
                if not duplicateFound and array.size(v1_bullObs) < 500
                    array.push(v1_bullObs, v1_OB.new(hl2[v1_pivotLen], low[v1_pivotLen], obTime, true, true))
                    
        if v1_bearCond and not na(high[v1_pivotLen])
            v1_lastBearOB := high[v1_pivotLen]
            bool duplicateFound = false
            int obTime = time[v1_pivotLen]
            if not na(obTime)
                for existing in v1_bearObs
                    if not na(existing) and existing.startTime == obTime
                        duplicateFound := true
                        break
                if not duplicateFound and array.size(v1_bearObs) < 500
                    array.push(v1_bearObs, v1_OB.new(high[v1_pivotLen], hl2[v1_pivotLen], obTime, false, true))
                    
        [v1_lastBullOB, v1_lastBearOB, v1_os, v1_bullCond, v1_bearCond, v1_bullCond, v1_bearCond]

v1_invalidate() =>
    float target_bull = v1_mitigation == "Close" ? ta.lowest(close, v1_pivotLen) : ta.lowest(v1_pivotLen)
    float target_bear = v1_mitigation == "Close" ? ta.highest(close, v1_pivotLen) : ta.highest(v1_pivotLen)
    
    if array.size(v1_bullObs) > 0
        int idx = array.size(v1_bullObs) - 1
        while idx >= 0
            ob = array.get(v1_bullObs, idx)
            if not na(ob) and ob.isValid
                if target_bull < ob.bottom
                    ob.isValid := false
                    array.set(v1_bullObs, idx, ob)
            idx -= 1
    if array.size(v1_bearObs) > 0
        int idx = array.size(v1_bearObs) - 1        
        while idx >= 0
            ob = array.get(v1_bearObs, idx)
            if not na(ob) and ob.isValid
                if target_bear > ob.top
                    ob.isValid := false
                    array.set(v1_bearObs, idx, ob)
            idx -= 1

v1_cleanup() =>
    if array.size(v1_bullObs) > 0
        for idx = array.size(v1_bullObs) - 1 to 0
            ob = array.get(v1_bullObs, idx)
            if na(ob) or not ob.isValid
                array.remove(v1_bullObs, idx)
        while array.size(v1_bullObs) > v1_bullShowLast * 2 and array.size(v1_bullObs) > 0
            array.pop(v1_bullObs)
            
    if array.size(v1_bearObs) > 0
        for idx = array.size(v1_bearObs) - 1 to 0
            ob = array.get(v1_bearObs, idx)
            if na(ob) or not ob.isValid
                array.remove(v1_bearObs, idx)
        while array.size(v1_bearObs) > v1_bearShowLast * 2 and array.size(v1_bearObs) > 0
            array.pop(v1_bearObs)

    if enableEngine1
        if array.size(v1_bullObs) > 0
            for i = 0 to math.min(v1_bullShowLast - 1, array.size(v1_bullObs) - 1)
                ob = array.get(v1_bullObs, i)
                if na(ob) or not ob.isValid or na(ob.startTime)
                    continue
                int rightTime = time_close
                if na(rightTime)
                    rightTime := time + (time - time[1])
                if not na(ob.startTime) and not na(ob.top) and not na(ob.bottom)
                    bx = box.new(ob.startTime, ob.top, rightTime, ob.bottom,
                        xloc=xloc.bar_time, bgcolor=v1_bullBgColor, 
                        border_color=v1_bullBorderColor, extend=extend.right)
                    array.push(v1_boxes, bx)
                if showMidLine
                    float midPrice = (ob.top + ob.bottom) / 2
                    if not na(ob.startTime) and not na(midPrice)
                        ln = line.new(ob.startTime, midPrice, rightTime, midPrice,
                            xloc=xloc.bar_time, color=v1_avgLineColor, 
                            style=getLineStyle(midLineStyle), width=1, extend=extend.right)
                        array.push(v1_lines, ln)
        if array.size(v1_bearObs) > 0
            for i = 0 to math.min(v1_bearShowLast - 1, array.size(v1_bearObs) - 1)

fx_findMostTouchedPrice(int fromIdx, int toIdx, int nBins) =>
    span = toIdx - fromIdx
    if span <= 0 or na(fromIdx) or na(toIdx) or fromIdx < 0 or toIdx > bar_index
        [float(na), int(0)]
    else
        minP = 1e10
        maxP = -1e10
        for idx = fromIdx to toIdx
            rel = bar_index - idx
            if rel >= 0 and rel <= bar_index
                lo = low[rel]
                hi = high[rel]
                if not na(lo) and not na(hi)
                    if lo < minP
                        minP := lo
                    if hi > maxP
                        maxP := hi
        if not (minP < maxP)
            [float(na), int(0)]
        else
            step = (maxP - minP) / nBins
            step := step <= 0 ? syminfo.mintick : step
            diffCnt = array.new<float>(nBins, 0.0)
            for idx = fromIdx to toIdx
                rel2 = bar_index - idx
                if rel2 >= 0
                    lo2 = low[rel2]
                    hi2 = high[rel2]
                    if not na(lo2) and not na(hi2)
                        sBin = int(math.floor((lo2 - minP) / step))
                        eBin = int(math.floor((hi2 - minP) / step))
                        sBin := sBin < 0 ? 0 : sBin > nBins - 1 ? nBins - 1 : sBin
                        eBin := eBin < 0 ? 0 : eBin > nBins - 1 ? nBins - 1 : eBin
                        if sBin < array.size(diffCnt)
                            array.set(diffCnt, sBin, array.get(diffCnt, sBin) + 1.0)
                        if eBin + 1 < nBins and eBin + 1 < array.size(diffCnt)
                            array.set(diffCnt, eBin + 1, array.get(diffCnt, eBin + 1) - 1.0)
            bestBin = 0
            bestCnt = 0.0
            runCnt = 0.0
            for i = 0 to nBins - 1
                if i < array.size(diffCnt)
                    runCnt += array.get(diffCnt, i)
                if runCnt > bestCnt
                    bestCnt := runCnt
                    bestBin := i
            poc = minP + (bestBin + 0.5) * step
            [poc, int(bestCnt)]

fx_volumeAtPrice(int fromIdx, int toIdx, float poc, string ltfVdTf) =>
    touches = 0
    totVol = 0.0
    bullVol = 0.0
    bearVol = 0.0
    span = toIdx - fromIdx
    if not na(poc) and span >= 0 and fromIdx >= 0 and toIdx <= bar_index
        for step = 0 to span
            idx = toIdx - step
            rel = bar_index - idx
            if rel >= 0 and rel <= bar_index
                lo = low[rel]
                hi = high[rel]
                if not na(lo) and not na(hi) and lo <= poc and hi >= poc
                    touches += 1
                    totVol += volume[rel]
                    bullVol += close[rel] > open[rel] ? volume[rel] : 0.0
                    bearVol += close[rel] < open[rel] ? volume[rel] : 0.0
    [touches, totVol, bullVol, bearVol]

fx_detect() =>
    var float local_fx_shPrice = na
    var int local_fx_shIdx = na
    var float local_fx_slPrice = na
    var int local_fx_slIdx = na
    
    if not na(fx_global_ph)
        local_fx_shPrice := fx_global_ph
        local_fx_shIdx := bar_index - fx_swingLen
    if not na(fx_global_pl)
        local_fx_slPrice := fx_global_pl
        local_fx_slIdx := bar_index - fx_swingLen
    
    if bar_index >= fx_swingLen * 2
        bool bosBear = not na(local_fx_slPrice) and bar_index > local_fx_slIdx and close < local_fx_slPrice and nz(close[1], close) >= local_fx_slPrice
        bool bosBull = not na(local_fx_shPrice) and bar_index > local_fx_shIdx and close > local_fx_shPrice and nz(close[1], close) <= local_fx_shPrice
        
        if bosBear
            int bosIdx = bar_index - 1
            int fromIdx = local_fx_slIdx
            if not na(fromIdx) and fromIdx >= 0 and fromIdx <= bar_index and bosIdx > fromIdx
                [poc, touches] = fx_findMostTouchedPrice(fromIdx, bosIdx, fx_pocBins)
                if not na(poc) and touches > 0
                    int bestIdx = -1
                    float runMaxHigh = na
                    for step = 0 to bosIdx - fromIdx
                        int idx = bosIdx - step
                        int rel = bar_index - idx
                        if rel >= 0 and rel <= bar_index and rel < bar_index
                            float lo = low[rel]
                            float hi = high[rel]
                            if not na(lo) and not na(hi)
                                runMaxHigh := na(runMaxHigh) ? hi : math.max(runMaxHigh, hi)
                                if lo <= poc and hi >= poc and hi == runMaxHigh
                                    bestIdx := idx
                    if bestIdx >= 0 and bestIdx <= bar_index
                        if not hasGapBetween(bestIdx, bosIdx, false)
                            int relBest = bar_index - bestIdx
                            if relBest >= 0 and relBest < bar_index
                                float top = high[relBest]
                                float bottom = low[relBest]
                                float atrVal = nz(fx_atr10, syminfo.mintick * 10)
                                if not na(top) and not na(bottom) and (top - bottom) <= atrVal * fx_maxATRMult
                                    if not obOverlapsActive(top, bottom, fx_obs)
                                        [touchesVol, totVol, bullVol, bearVol] = fx_volumeAtPrice(fromIdx, bosIdx, poc, fx_vdTfIn)
                                        int obTime = time[relBest]
                                        if not na(obTime) and totVol > 0
                                            array.push(fx_obs, fx_OB.new(top, bottom, obTime, totVol, bullVol, bearVol, true, false, na))
        
        if bosBull
            int bosIdx2 = bar_index - 1
            int fromIdx2 = local_fx_shIdx
            if not na(fromIdx2) and fromIdx2 >= 0 and fromIdx2 <= bar_index and bosIdx2 > fromIdx2
                [poc2, touches2] = fx_findMostTouchedPrice(fromIdx2, bosIdx2, fx_pocBins)
                if not na(poc2) and touches2 > 0
                    int bestIdx2 = -1
                    float runMinLow = na
                    for step = 0 to bosIdx2 - fromIdx2
                        int idx2 = bosIdx2 - step
                        int rel2 = bar_index - idx2
                        if rel2 >= 0 and rel2 <= bar_index and rel2 < bar_index
                            float lo2 = low[rel2]
                            float hi2 = high[rel2]
                            if not na(lo2) and not na(hi2)
                                runMinLow := na(runMinLow) ? lo2 : math.min(runMinLow, lo2)
                                if lo2 <= poc2 and hi2 >= poc2 and lo2 == runMinLow
                                    bestIdx2 := idx2
                    if bestIdx2 >= 0 and bestIdx2 <= bar_index
                        if not hasGapBetween(bestIdx2, bosIdx2, true)
                            int relBest2 = bar_index - bestIdx2
                            if relBest2 >= 0 and relBest2 < bar_index
                                float top2 = high[relBest2]
                                float bottom2 = low[relBest2]
                                float atrVal = nz(fx_atr10, syminfo.mintick * 10)
                                if not na(top2) and not na(bottom2) and (top2 - bottom2) <= atrVal * fx_maxATRMult
                                    if not obOverlapsActive(top2, bottom2, fx_obs)
                                        [touchesVol2, totVol2, bullVol2, bearVol2] = fx_volumeAtPrice(fromIdx2, bosIdx2, poc2, fx_vdTfIn)
                                        int obTime2 = time[relBest2]
                                        if not na(obTime2) and totVol2 > 0
                                            array.push(fx_obs, fx_OB.new(top2, bottom2, obTime2, totVol2, bullVol2, bearVol2, true, false, na))
    pruneObs(fx_obs)

                    if close < ob.bottom or low < ob.bottom
                        ob.isBreaker := true
                        ob.breakTime := time
                        ob.isValid := false
                        array.set(obs, idx, ob)
                else

                        ob.isBreaker := true
                        ob.breakTime := time
                        ob.isValid := false

    obs

fx_checkRetests(array<fx_OB> obs) =>
    var array<fx_RetestRec> local_fx_retests = array.new<fx_RetestRec>()
    
    if bar_index >= 1 and array.size(obs) > 0
        for i = 0 to array.size(obs) - 1
            ob = array.get(obs, i)
            if not na(ob) and ob.isValid and not ob.retested
                bool retestPrev = false
                if not na(ob.bullVol) and not na(ob.bearVol) and ob.bullVol >= ob.bearVol
                    bool opensAbovePrev = open[1] > ob.top
                    bool closesAbovePrev = close[1] > ob.top
                    bool wickTouchesPrev = low[1] <= ob.top and low[1] >= ob.bottom
                    retestPrev := opensAbovePrev and closesAbovePrev and wickTouchesPrev
                else if not na(ob.bullVol) and not na(ob.bearVol)
                    bool opensBelowPrev = open[1] < ob.bottom
                    bool closesBelowPrev = close[1] < ob.bottom
                    bool wickTouchesPrev = high[1] >= ob.bottom and high[1] <= ob.top
                    retestPrev := opensBelowPrev and closesBelowPrev and wickTouchesPrev
                
                if retestPrev
                    ob.retested := true
                    array.set(obs, i, ob)
                    array.unshift(local_fx_retests, fx_RetestRec.new(bar_index - 1, ob.bullVol >= ob.bearVol, ob.startTime, ob.startTime, ob.startTime))
    local_fx_retests

ua_calcStrengths(int barIndex) =>
    bullStr = 0.0
    bearStr = 0.0
    for idx = 0 to ua_swingLen - 1
        if bar_index - idx >= 0
            if open[idx] > close[idx]
                bearStr += volume[idx]
            else
                bullStr += volume[idx]
    [bullStr, bearStr]

ua_isOverlapping(ua_OB newBlock, array<ua_OB> existingArray) =>
    if na(newBlock)
        false
    else
        bool overlapping = false
        for existing in existingArray
            if na(existing) or not existing.isValid
                continue
            if (newBlock.top >= existing.bottom and newBlock.bottom <= existing.top) or (existing.top >= newBlock.bottom and existing.bottom <= newBlock.top)
                overlapping := true
                break
        overlapping

ua_detect() =>
    if not na(ua_global_pl) and bar_index >= ua_swingLen
        int highestIdx = -1
        float highestHigh = na
        for i = 1 to ua_swingLen
            if bar_index - i >= 0 and not na(high[i]) and (na(highestHigh) or high[i] > highestHigh)
                highestHigh := high[i]
                highestIdx := i
        if highestIdx > 0
            float top = high[highestIdx]
            float bottom = low[highestIdx]
            [bullStr1, bearStr1] = ua_calcStrengths(highestIdx)
            float totalStr = bullStr1 + bearStr1
            float bullPct = totalStr > 0 ? bullStr1 / totalStr * 100 : 50
            float bearPct = totalStr > 0 ? bearStr1 / totalStr * 100 : 50
            int obStart = time[highestIdx]
            if not na(obStart)
                bool duplicateFound = false
                if ua_hideOverlap == "True"
                    duplicateFound := ua_isOverlapping(ua_OB.new(top, bottom, obStart, bullPct, bearPct, true), ua_obs)
                if not duplicateFound and array.size(ua_obs) < 500
                    array.push(ua_obs, ua_OB.new(top, bottom, obStart, bullPct, bearPct, true))
    
    if not na(ua_global_ph) and bar_index >= ua_swingLen
        int lowestIdx = -1
        float lowestLow = na

zi_volumePower(float v, float c, float l, float h, bool isBull) =>
    float rangeVal = h - l
    if rangeVal == 0 or na(rangeVal) or v == 0 or v <= 0.0000001 or na(c) or na(l) or na(h)
        0.0
    else
        float raw = v * (isBull ? (c - l) : (h - c)) / rangeVal
        if raw <= 0
            0.0
        else
            math.min(100.0, raw / v * 100.0)

    float reqCloseBack = request.security(syminfo.tickerid, zi_tf, close[safeOffset], ignore_invalid_symbol=true, lookahead=barmerge.lookahead_on)
    float reqHighBack = request.security(syminfo.tickerid, zi_tf, high[safeOffset], ignore_invalid_symbol=true, lookahead=barmerge.lookahead_on)

    bool sumBear_val = math.sum(close < open ? 1 : 0, safeLookback) == safeLookback

bb_detect() =>
    var int local_bb_crossIndex = na
    var bool local_bb_ob_created = false
    var bool local_bb_ob_created_bull = false
    var int local_bb_last_ob_bar = na
    
    [req_open, req_high, req_low, req_close] = request.security(syminfo.tickerid, bb_tf, [open, high, low, close], ignore_invalid_symbol=true)
    
    float pc_mtf_local = 0.0
    float open4 = nz(open[4], 0)
    if open4 != 0
        pc_mtf_local := (open - open4) / math.abs(open4) * 100
    else if not na(open[4]) and open[4] != 0
        pc_mtf_local := (open - open[4]) / math.abs(open[4]) * 100
    
    bool local_crossunder = ta.crossunder(pc_mtf_local, -bb_sens)
    bool local_crossover = ta.crossover(pc_mtf_local, bb_sens)
    
    if bar_index >= 4
        if local_crossunder
            local_bb_ob_created := true
            local_bb_crossIndex := bar_index
        if local_crossover
            local_bb_ob_created_bull := true
            local_bb_crossIndex := bar_index
        
        if local_bb_ob_created and (na(local_bb_last_ob_bar) or local_bb_crossIndex - local_bb_last_ob_bar > 5)
            for i = 4 to 15
                if bar_index - i >= 0 and not na(req_close[i]) and not na(req_open[i]) and not na(req_high[i]) and not na(req_low[i])
                    if req_close[i] > req_open[i]

                        if not na(obTime)
                            array.push(bb_obs, bb_OB.new(req_high[i], req_low[i], obTime, false, true))
                            local_bb_last_ob_bar := local_bb_crossIndex
                            local_bb_ob_created := false
                        break
            local_bb_ob_created := false
        
        if local_bb_ob_created_bull and (na(local_bb_last_ob_bar) or local_bb_crossIndex - local_bb_last_ob_bar > 5)
            for i = 4 to 15
                if bar_index - i >= 0 and not na(req_close[i]) and not na(req_open[i]) and not na(req_high[i]) and not na(req_low[i])
